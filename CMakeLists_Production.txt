# ============================================================================
# DBX4 PRODUCTION BUILD - ALL MODULES
# ============================================================================

cmake_minimum_required(VERSION 3.20)
project(DBX4 VERSION 2.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(MSVC)
    add_compile_options(/MP /W4 /O2)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# ============================================================================
# PHASE 1: STORAGE ENGINE (PRODUCTION)
# ============================================================================

add_executable(storage_production src/storage_engine_production.cpp)
set_target_properties(storage_production PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>
)

# ============================================================================
# QUERY EXECUTOR
# ============================================================================

add_executable(query_executor src/query_executor.cpp)
set_target_properties(query_executor PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>
)

# ============================================================================
# TRANSACTION MANAGER
# ============================================================================

add_executable(transaction_manager src/transaction_manager.cpp)
set_target_properties(transaction_manager PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>
)

# ============================================================================
# PHASE 2-6: EXISTING MODULES
# ============================================================================

add_executable(storage_complete src/storage_complete.cpp)
add_executable(mvcc_locks src/mvcc_locks.cpp)
add_executable(recovery_wal src/recovery_wal.cpp)
add_executable(indexing src/indexing.cpp)
add_executable(advanced_features src/advanced_features.cpp)
add_executable(testing_suite src/testing_suite.cpp)

foreach(target storage_complete mvcc_locks recovery_wal indexing advanced_features testing_suite)
    set_target_properties(${target} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>
    )
endforeach()

# ============================================================================
# TESTING
# ============================================================================

enable_testing()

add_test(NAME Storage_Production COMMAND storage_production WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)
add_test(NAME Query_Executor COMMAND query_executor WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)
add_test(NAME Transaction_Manager COMMAND transaction_manager WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)
add_test(NAME Phase1_StorageComplete COMMAND storage_complete WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)
add_test(NAME Phase2_MVCCLocks COMMAND mvcc_locks WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)
add_test(NAME Phase3_RecoveryWAL COMMAND recovery_wal WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)
add_test(NAME Phase4_Indexing COMMAND indexing WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)
add_test(NAME Phase5_AdvancedFeatures COMMAND advanced_features WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)
add_test(NAME Phase6_TestingSuite COMMAND testing_suite WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/$<CONFIG>)

message(STATUS "")
message(STATUS "=====================================")
message(STATUS "DBX4 PRODUCTION BUILD")
message(STATUS "=====================================")
message(STATUS "Modules: 9 executables")
message(STATUS "Total LOC: 5,494+")
message(STATUS "Tests: 11 test suites")
message(STATUS "")
message(STATUS "Build with: cmake --build . --config Release -j8")
message(STATUS "Run tests: ctest -C Release -V")
message(STATUS "=====================================")
message(STATUS "")
